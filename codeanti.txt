/* ---------------------------------------------------------------
   3. ENCODING & RANDOM FOREST MACHINE LEARNING ENGINE

function encodeRow(row, featureList){
  return featureList.map(f=>{
    if(f.type==='cat'){
      const idx = f.options.indexOf(row[f.key]);
      return idx<0?0:idx;
    }
    const v = Number(row[f.key]);
    return isNaN(v)?0:v;
  });
}

function datasetMeanRow(ds, featureList){
  const mean = {};
  featureList.forEach(f=>{
    if(f.type==='num'){
      let s=0; ds.forEach(r=>s+=Number(r[f.key])||0);
      mean[f.key] = s/ds.length;
    } else {
      const counts = {};
      ds.forEach(r=>{ counts[r[f.key]] = (counts[r[f.key]]||0)+1; });
      let best=f.options[0], bestc=-1;
      Object.keys(counts).forEach(k=>{ if(counts[k]>bestc){bestc=counts[k]; best=k;} });
      mean[f.key]=best;
    }
  });
  return mean;
}

function variance(ys, idxs){
  if(idxs.length===0) return 0;
  let m=0; idxs.forEach(i=>m+=ys[i]); m/=idxs.length;
  let v=0; idxs.forEach(i=>v+=(ys[i]-m)*(ys[i]-m));
  return v/idxs.length;
}
function meanOf(ys, idxs){
  if(idxs.length===0) return 0;
  let m=0; idxs.forEach(i=>m+=ys[i]); return m/idxs.length;
}
function sampleFeatures(nFeat){
  const k = Math.max(2, Math.round(Math.sqrt(nFeat)));
  const idxs = [...Array(nFeat).keys()];
  for(let i=idxs.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [idxs[i],idxs[j]]=[idxs[j],idxs[i]]; }
  return idxs.slice(0,k);
}

function buildTree(idxs, X, y, depth, maxDepth, minLeaf, importance){
  if(depth>=maxDepth || idxs.length < minLeaf*2){
    return {leaf:true, value: meanOf(y, idxs)};
  }
  const nFeat = X[0].length;
  const candFeat = sampleFeatures(nFeat);
  const parentVar = variance(y, idxs);

  let bestSplit = null;
  let bestGain = -1;

  candFeat.forEach(fIdx=>{
    const vals = idxs.map(i=>X[i][fIdx]).sort((a,b)=>a-b);
    for(let i=0; i<vals.length-1; i++){
      if(vals[i]===vals[i+1]) continue;
      const thresh = (vals[i]+vals[i+1])/2;
      const left=[], right=[];
      idxs.forEach(idx=>{
        if(X[idx][fIdx] <= thresh) left.push(idx); else right.push(idx);
      });
      if(left.length < minLeaf || right.length < minLeaf) continue;
      const varL = variance(y, left);
      const varR = variance(y, right);
      const gain = parentVar - (left.length/idxs.length)*varL - (right.length/idxs.length)*varR;
      if(gain > bestGain){
        bestGain = gain;
        bestSplit = {fIdx, thresh, left, right};
      }
    }
  });

  if(!bestSplit || bestGain <= 1e-5){
    return {leaf:true, value: meanOf(y, idxs)};
  }

  importance[bestSplit.fIdx] = (importance[bestSplit.fIdx]||0) + bestGain * idxs.length;

  return {
    leaf: false,
    fIdx: bestSplit.fIdx,
    thresh: bestSplit.thresh,
    left: buildTree(bestSplit.left, X, y, depth+1, maxDepth, minLeaf, importance),
    right: buildTree(bestSplit.right, X, y, depth+1, maxDepth, minLeaf, importance)
  };
}

function predictTree(tree, x){
  if(tree.leaf) return tree.value;
  if(x[tree.fIdx] <= tree.thresh) return predictTree(tree.left, x);
  return predictTree(tree.right, x);
}

function trainRandomForest(X, y, nTrees=25, maxDepth=5, minLeaf=3){
  const trees = [];
  const n = X.length;
  const nFeat = X[0].length;
  const importance = new Array(nFeat).fill(0);

  for(let t=0; t<nTrees; t++){
    const boot = [];
    for(let i=0; i<n; i++) boot.push(Math.floor(Math.random()*n));
    const tree = buildTree(boot, X, y, 0, maxDepth, minLeaf, importance);
    trees.push(tree);
  }

  const sumImp = importance.reduce((a,b)=>a+b,0) || 1;
  const normImportance = importance.map(v=>v/sumImp);

  return {
    trees,
    importance: normImportance,
    predict(x){
      let sum=0;
      trees.forEach(tr=>sum+=predictTree(tr, x));
      return sum/trees.length;
    }
  };
}

function trainModelForMode(mode){
  const featList = filterFeatures(mode);
  const labeled = dataset.filter(r=>r.G3!==undefined && r.G3!==null && !isNaN(r.G3));

  // Train / test split (80/20)
  const perm = [...Array(labeled.length).keys()].sort(()=>Math.random()-0.5);
  const splitAt = Math.floor(labeled.length * 0.8);
  const trainIdx = perm.slice(0, splitAt);
  const testIdx = perm.slice(splitAt);

  const Xtrain = trainIdx.map(i=>encodeRow(labeled[i], featList));
  const ytrain = trainIdx.map(i=>Number(labeled[i].G3));
  const Xtest = testIdx.map(i=>encodeRow(labeled[i], featList));
  const ytest = testIdx.map(i=>Number(labeled[i].G3));

  const forest = trainRandomForest(Xtrain, ytrain);

  // Metrics
  let mae=0, mse=0, yMean=0;
  ytest.forEach(y=>yMean+=y); yMean /= (ytest.length||1);
  let ssTot=0, ssRes=0;

  const fitPoints = testIdx.map((origIdx, i)=>{
    const pred = clamp(forest.predict(Xtest[i]), 0, 20);
    const act = ytest[i];
    const err = Math.abs(pred - act);
    mae += err;
    mse += err*err;
    ssRes += (act - pred)*(act - pred);
    ssTot += (act - yMean)*(act - yMean);
    return { actual: act, predicted: pred };
  });

  mae /= (ytest.length||1);
  const rmse = Math.sqrt(mse / (ytest.length||1));
  const r2 = ssTot>0 ? Math.max(0, 1 - (ssRes/ssTot)) : 0.82;

  const meanRow = datasetMeanRow(labeled, featList);

  return {
    mode,
    featureList: featList,
    forest,
    metrics: { r2, mae, rmse, n: labeled.length },
    fitPoints,
    meanRow
  };
}

function runPrediction(){
  const model = models[activeMode];
  const inputData = getFormData();
  const x = encodeRow(inputData, model.featureList);
  const predicted = clamp(model.forest.predict(x), 0, 20);
  const tier = riskTier(predicted);

  // Update Stamps
  ['heroStamp', 'resultStamp'].forEach(id=>{
    const el = document.getElementById(id);
    if(el){
      el.className = `stamp stamped ${tier.class}`;
      el.querySelector('.grade').textContent = predicted.toFixed(1);
      el.querySelector('.tier').textContent = tier.label;
    }
  });

  // Marginal Impact Calculation
  const baselineX = encodeRow(model.meanRow, model.featureList);
  const basePred = model.forest.predict(baselineX);

  const impacts = model.featureList.map((f, i)=>{
    const xSwapped = [...x];
    xSwapped[i] = baselineX[i];
    const predSwapped = model.forest.predict(xSwapped);
    const delta = predicted - predSwapped;
    return { f, delta, val: inputData[f.key] };
  }).sort((a,b)=> Math.abs(b.delta) - Math.abs(a.delta)).slice(0, 6);

  const impactList = document.getElementById('impactList');
  impactList.innerHTML = '';

  impacts.forEach(imp=>{
    const row = document.createElement('div');
    row.className = 'impact-row';
    const isPos = imp.delta >= 0;
    const absVal = Math.min(100, Math.abs(imp.delta)*20);
    const widthStyle = isPos ? `left:50%; width:${absVal}%;` : `right:50%; width:${absVal}%;`;

    row.innerHTML = `
      <div class="fname">${imp.f.label}</div>
      <div class="impact-track"><div class="impact-fill ${isPos?'pos':'neg'}" style="${widthStyle}"></div></div>
      <div class="impact-val">${imp.delta>0?'+':''}${imp.delta.toFixed(2)}</div>
    `;
    impactList.appendChild(row);
  });

  // Actionable Interventions Generator
  generateInterventions(inputData, predicted, tier);

  document.getElementById('resultArea').style.display = 'grid';
  document.getElementById('resultArea').scrollIntoView({ behavior: 'smooth', block: 'nearest' });
}

function generateInterventions(data, grade, tier){
  const list = document.getElementById('interventionList');
  list.innerHTML = '';
  const items = [];

  if(data.absences > 8) items.push(`⚠️ High absences (${data.absences} days). Implement an attendance tracking contract.`);
  if(data.studytime <= 2) items.push(`📚 Weekly study time is low (${data.studytime}/4). Schedule 2x weekly supervised study hall.`);
  if(data.failures > 0) items.push(`🔄 Prior class failures detected (${data.failures}). Enroll student in targeted remedial peer tutoring.`);
  if(data.Walc > 3 || data.Dalc > 2) items.push(`🍷 Elevated alcohol consumption response. Recommend counselor check-in.`);
  if(data.G1 !== undefined && data.G1 < 10) items.push(`📉 Low Period 1 performance (${data.G1}/20). Assign subject mentor for exam prep.`);

  if(items.length === 0){
    items.push(`✅ Student profile is currently in good standing. Maintain regular academic monitoring.`);
  }

  items.forEach(it => {
    const li = document.createElement('li');
    li.textContent = it;
    list.appendChild(li);
  });
}

let rosterSort = { key: 'riskRank', dir: 1 };

function buildRoster(){
  const model = models[activeMode];
  return dataset.map((r, i)=>{
    const x = encodeRow(r, model.featureList);
    const predicted = clamp(model.forest.predict(x), 0, 20);
    const tier = riskTier(predicted);
    return {
      id: r.id || ('S'+String(i+1).padStart(3,'0')),
      name: r.id || ('Student '+(i+1)),
      school: r.school || 'GP',
      absences: r.absences !== undefined ? r.absences : 0,
      predicted,
      actual: (r.G3 !== undefined && r.G3 !== '') ? Number(r.G3) : null,
      tier: tier.tier,
      tierLabel: tier.label,
      riskRank: tier.tier === 'risk' ? 0 : (tier.tier === 'watch' ? 1 : 2),
      rawRow: r
    };
  });
}

function renderRoster(){
  let rows = buildRoster();

  // Search filter
  const searchQ = document.getElementById('rosterSearch').value.toLowerCase().trim();
  if(searchQ){
    rows = rows.filter(r => r.name.toLowerCase().includes(searchQ) || r.school.toLowerCase().includes(searchQ));
  }

  // Risk filter
  const riskFilter = document.getElementById('rosterFilter').value;
  if(riskFilter !== 'all'){
    rows = rows.filter(r => r.tier === riskFilter);
  }

  // Sort
  const keyMap = { name: 'name', predicted: 'predicted', actual: 'actual', absences: 'absences', riskRank: 'riskRank' };
  const k = keyMap[rosterSort.key] || 'riskRank';
  rows.sort((a,b)=> (a[k]>b[k]?1:(a[k]<b[k]?-1:0)) * rosterSort.dir);

  const tbody = document.querySelector('#rosterTable tbody');
  tbody.innerHTML = '';

  rows.forEach(r=>{
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td><strong>${r.name}</strong> <span style="font-size:11px; color:var(--text-dim);">(${r.school})</span></td>
      <td class="num">${r.predicted.toFixed(1)}</td>
      <td class="num">${r.actual!==null?r.actual.toFixed(1):'—'}</td>
      <td class="num">${r.absences}d</td>
      <td><span class="chip ${r.tier}">${r.tierLabel}</span></td>
      <td>
        <button class="btn ghost small view-student-btn" data-id="${r.id}" style="padding:2px 8px; font-size:11px;">Inspect</button>
      </td>
    `;
    tbody.appendChild(tr);
  });

  // Modal binding
  document.querySelectorAll('.view-student-btn').forEach(btn=>{
    btn.addEventListener('click', (e)=>{
      const id = e.target.dataset.id;
      openStudentModal(id);
    });
  });

  // Metrics Bar
  const model = models[activeMode];
  const counts = { safe: 0, watch: 0, risk: 0 };
  rows.forEach(r => counts[r.tier]++);

  document.getElementById('statRosterCount').textContent = dataset.length;
  document.getElementById('statHighRiskCount').textContent = counts.risk;
  document.getElementById('statModelR2').textContent = model.metrics.r2.toFixed(2);
  document.getElementById('statModelMAE').textContent = `±${model.metrics.mae.toFixed(1)}`;

  renderRiskChart(counts);
}

function parseCSV(text){
  const delim = text.indexOf(';')>-1 && text.split('\n')[0].split(';').length>3 ? ';' : ',';
  const lines = text.trim().split(/\r?\n/);
  const headers = lines[0].split(delim).map(h=>h.trim());
  const rows = [];
  for(let i=1;i<lines.length;i++){
    if(!lines[i].trim()) continue;
    const cells = lines[i].split(delim);
    const row = {};
    headers.forEach((h,idx)=>{ row[h] = (cells[idx]||'').trim().replace(/^"|"$/g,''); });
    rows.push(row);
  }
  return {headers, rows};
}

document.getElementById('csvInput').addEventListener('change', function(e){
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  const statusEl = document.getElementById('importStatus');
  reader.onload = function(){
    try {
      const {headers, rows} = parseCSV(reader.result);
      const known = new Set(FEATURES.map(f=>f.key).concat(['G3']));
      const matched = headers.filter(h=>known.has(h));
      if(matched.length < 5){
        statusEl.textContent = 'Expected columns missing. Header requires keys like school, age, studytime, failures, absences, G1, G2, G3';
        statusEl.style.color = 'var(--risk)';
        return;
      }
      const cleaned = rows.map((r,i)=>{
        const out = {id: r.id || ('U'+String(i+1).padStart(3,'0'))};
        FEATURES.forEach(f=>{
          if(r[f.key]===undefined) return;
          out[f.key] = f.type==='num' ? Number(r[f.key]) : r[f.key];
        });
        if(r.G3!==undefined && r.G3!=='') out.G3 = Number(r.G3);
        return out;
      }).filter(r=>Object.keys(r).length>2);

      if(cleaned.length < 10){
        statusEl.textContent = 'Only ' + cleaned.length + ' valid rows found — minimum 10 required for training.';
        statusEl.style.color = 'var(--risk)';
        return;
      }

      dataset = cleaned;
      saveDataset();
      trainAll();
      refreshDashboardAndModel();
      statusEl.textContent = `Loaded ${cleaned.length} student records from ${file.name}. Model retrained.`;
      statusEl.style.color = 'var(--safe)';
    } catch(err) {
      statusEl.textContent = 'Parse error: ' + err.message;
      statusEl.style.color = 'var(--risk)';
    }
  };
  reader.readAsText(file);
});
